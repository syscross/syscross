
cmake_minimum_required(VERSION 4.0)

project(cmake)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_compile_definitions(
    UNICODE
    _UNICODE
    NOMINMAX
    WIN32_LEAN_AND_MEAN
)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/bin")

set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_DEBUG "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/lib")

file(GLOB_RECURSE LIB_SRC "lib/src/*")

add_library(LIB STATIC ${LIB_SRC})

target_include_directories(LIB
    INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/inc"
    PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/lib/inc"
)

file(GLOB_RECURSE BIN_SRC "bin/*")

foreach(BIN ${BIN_SRC})
    get_filename_component(BIN_NAME ${BIN} NAME_WLE)
    add_executable(${BIN_NAME} WIN32 ${BIN})
    target_link_libraries(${BIN_NAME} PRIVATE LIB)
endforeach()
